    function validateLegalDocSelect() {
        const legalDocSelect = document.getElementById('legalDocSelect');
        const isLegalDocEnabled = !legalDocSelect.disabled;
        const legalDocValue = legalDocSelect.value.trim();
        const errorMsg = document.querySelector('.error-message');

        if (isLegalDocEnabled && legalDocValue === '') {
            // Show the error message
            errorMsg.textContent = "Please select a Legal Name Verification Document.";
            errorMsg.style.display = "block";
            return false;
        } else {
            // Hide the error message if validation passes
            errorMsg.style.display = "none";
            return true;
        }
    }
